cmake_minimum_required(VERSION 3.16)
# macOS: set before project(). Universal (arm64 + x86_64) release builds pass
# -DCMAKE_OSX_ARCHITECTURES="arm64;x86_64" and need the fetched SDL3.
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
project(SealTeam VERSION 0.5.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()

# --- SDL3: use an installed package if there is one, otherwise build it from source.
option(SEALTEAM_FETCH_SDL3 "Download and build SDL3 if no installed SDL3 is found" ON)
find_package(SDL3 CONFIG QUIET)
if(NOT SDL3_FOUND)
    if(NOT SEALTEAM_FETCH_SDL3)
        message(FATAL_ERROR "SDL3 not found. Install it or enable SEALTEAM_FETCH_SDL3.")
    endif()
    message(STATUS "SDL3 not found - fetching and building it from source")
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3
        GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
        GIT_TAG release-3.2.28
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(SDL3)
endif()

# --- ymfm: Yamaha FM (OPL2/OPL3) emulator, BSD-3-Clause, vendored in third_party/ymfm
add_library(ymfm STATIC
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party/ymfm/src/ymfm_opl.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party/ymfm/src/ymfm_adpcm.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/third_party/ymfm/src/ymfm_pcm.cpp)
target_include_directories(ymfm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/ymfm/src)
if(NOT MSVC)
    target_compile_options(ymfm PRIVATE -w)  # third-party code
endif()

# --- Main executable
file(GLOB_RECURSE SEALTEAM_SOURCES CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/*.h)
list(FILTER SEALTEAM_SOURCES EXCLUDE REGEX "/src/tools/")  # standalone test tools

# Application icon and version info (assets/icon, generated by tools/make_icon.py).
if(WIN32)
    enable_language(RC)
    list(APPEND SEALTEAM_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam.rc)
endif()

# macOS: SealTeam.app bundle with an .icns made from the 1024 px icon.
if(APPLE)
    set(SEALTEAM_ICNS ${CMAKE_CURRENT_BINARY_DIR}/sealteam.icns)
    set(SEALTEAM_ICONSET ${CMAKE_CURRENT_BINARY_DIR}/sealteam.iconset)
    add_custom_command(OUTPUT ${SEALTEAM_ICNS}
        COMMAND ${CMAKE_COMMAND} -E make_directory ${SEALTEAM_ICONSET}
        COMMAND sips -z 16 16 ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam_1024.png --out ${SEALTEAM_ICONSET}/icon_16x16.png
        COMMAND sips -z 32 32 ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam_1024.png --out ${SEALTEAM_ICONSET}/icon_16x16@2x.png
        COMMAND sips -z 32 32 ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam_1024.png --out ${SEALTEAM_ICONSET}/icon_32x32.png
        COMMAND sips -z 64 64 ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam_1024.png --out ${SEALTEAM_ICONSET}/icon_32x32@2x.png
        COMMAND sips -z 128 128 ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam_1024.png --out ${SEALTEAM_ICONSET}/icon_128x128.png
        COMMAND sips -z 256 256 ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam_1024.png --out ${SEALTEAM_ICONSET}/icon_128x128@2x.png
        COMMAND sips -z 256 256 ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam_1024.png --out ${SEALTEAM_ICONSET}/icon_256x256.png
        COMMAND sips -z 512 512 ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam_1024.png --out ${SEALTEAM_ICONSET}/icon_256x256@2x.png
        COMMAND sips -z 512 512 ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam_1024.png --out ${SEALTEAM_ICONSET}/icon_512x512.png
        COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam_1024.png ${SEALTEAM_ICONSET}/icon_512x512@2x.png
        COMMAND iconutil -c icns ${SEALTEAM_ICONSET} -o ${SEALTEAM_ICNS}
        DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/assets/icon/sealteam_1024.png
        VERBATIM)
    set_source_files_properties(${SEALTEAM_ICNS} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    list(APPEND SEALTEAM_SOURCES ${SEALTEAM_ICNS})
endif()

add_executable(sealteam WIN32 MACOSX_BUNDLE ${SEALTEAM_SOURCES})
if(APPLE)
    configure_file(${CMAKE_CURRENT_SOURCE_DIR}/assets/macos/Info.plist.in
                   ${CMAKE_CURRENT_BINARY_DIR}/Info.plist @ONLY)
    set_target_properties(sealteam PROPERTIES
        OUTPUT_NAME SealTeam
        MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_BINARY_DIR}/Info.plist)
endif()
# Keep console output for developer commands and logs on Windows too.
if(MSVC)
    target_link_options(sealteam PRIVATE /SUBSYSTEM:CONSOLE)
elseif(MINGW)
    target_link_options(sealteam PRIVATE -mconsole)
endif()
target_include_directories(sealteam PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(sealteam PRIVATE SDL3::SDL3)
target_link_libraries(sealteam PRIVATE ymfm)
if(WIN32)
    target_link_libraries(sealteam PRIVATE dbghelp)  # crash reports with symbols
endif()

if(MSVC)
    target_compile_options(sealteam PRIVATE /W4 /permissive-)
    target_compile_definitions(sealteam PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(sealteam PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
if(MINGW)
    # Self-contained exe: no libgcc/libstdc++/winpthread DLLs needed.
    target_link_options(sealteam PRIVATE -static-libgcc -static-libstdc++)
    if(NOT TARGET SDL3::SDL3-shared)
        target_link_options(sealteam PRIVATE -static)
    endif()
endif()

# A shared SDL3 (installed package) must sit next to the executable on Windows.
if(WIN32 AND TARGET SDL3::SDL3-shared)
    add_custom_command(TARGET sealteam POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:sealteam>)
endif()

# --- Audio test tool (offline XMIDI/VOC rendering; not part of the game)
file(GLOB SEALTEAM_AUDIO_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/audio/*.cpp)
add_executable(sealteam_audiotest
    ${CMAKE_CURRENT_SOURCE_DIR}/src/tools/audiotest.cpp
    ${SEALTEAM_AUDIO_SOURCES}
    ${CMAKE_CURRENT_SOURCE_DIR}/src/data/ealib.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/data/gamefs.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/src/core/log.cpp)
target_include_directories(sealteam_audiotest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(sealteam_audiotest PRIVATE ymfm SDL3::SDL3)
if(NOT MSVC)
    target_compile_options(sealteam_audiotest PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
if(MINGW)
    target_link_options(sealteam_audiotest PRIVATE -static-libgcc -static-libstdc++)
    if(NOT TARGET SDL3::SDL3-shared)
        target_link_options(sealteam_audiotest PRIVATE -static)
    endif()
endif()
